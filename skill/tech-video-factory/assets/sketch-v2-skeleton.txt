const PAGES = [
  {C: PageHook,  start: 0,    end: 330},   // 开场封面：第 0 帧完整静态，不描画
  {C: PageX1,    start: 330,  end: ...},   // 每页一个概念，边界=配音句窗口
  // ... 按句切页；单页内容帧 200~500（7~16s）；一页只干一件事（结尾 CTA 单独一页）
];

const PageTurn: React.FC<{children; off: number; len: number; first: boolean; last: boolean}> = ({
  children, off, len, first, last,
}) => {
  const frame = useCurrentFrame();
  const oIn = first ? 1 : interpolate(frame, [0, 10], [0, 1], clamp);
  const lf = frame - off;
  const oOut = last ? 1 : interpolate(lf, [len - 10, len], [1, 0], clamp);
  return <AbsoluteFill style={{opacity: Math.min(oIn, oOut)}}>{children}</AbsoluteFill>;
};

// 主组件里：
{PAGES.map(({C, start, end}, i) => {
  const first = i === 0, last = i === PAGES.length - 1;
  const len = end - start;
  const off = first ? 0 : 10;
  return (
    <Sequence key={i} from={first ? 0 : start - 10} durationInFrames={len + 12}>
      <PageTurn off={off} len={len} first={first} last={last}><C off={off} /></PageTurn>
    </Sequence>
  );
})}
// 页内取内容帧：const f = useCurrentFrame() - off;
